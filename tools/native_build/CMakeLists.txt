cmake_minimum_required(VERSION 3.24)
project(RoyaleNativeResearch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 3.0.1 CONFIG REQUIRED)
set(ENGINE "${CMAKE_CURRENT_LIST_DIR}/../../third_party/ClashRoyaleAi")
pybind11_add_module(clash_royale_env "${ENGINE}/src/bindings.cpp")
target_include_directories(clash_royale_env PRIVATE "${ENGINE}/include/core" "${ENGINE}/include/entities" "${ENGINE}/include/rendering")
if(MSVC)
 target_compile_options(clash_royale_env PRIVATE /bigobj /utf-8)
endif()
set_target_properties(clash_royale_env PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${ENGINE}/python_ai")
foreach(config RELEASE DEBUG RELWITHDEBINFO)
 set_target_properties(clash_royale_env PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${config} "${ENGINE}/python_ai")
endforeach()
option(BUILD_NATIVE_TESTS "Build upstream native tests using locally downloaded Catch2" ON)
if(BUILD_NATIVE_TESTS)
 file(GLOB TESTS "${ENGINE}/tests/core/*.cpp" "${ENGINE}/tests/entities/*.cpp")
 add_executable(ClashRoyaleTests ${TESTS} "${CMAKE_CURRENT_LIST_DIR}/../../.cache/native_catch2/catch_amalgamated.cpp")
 target_include_directories(ClashRoyaleTests PRIVATE "${ENGINE}/include/core" "${ENGINE}/include/entities" "${ENGINE}/include/rendering" "${ENGINE}/tests/entities" "${CMAKE_CURRENT_LIST_DIR}/../../.cache/native_catch2")
 if(MSVC)
  target_compile_options(ClashRoyaleTests PRIVATE /bigobj /utf-8)
 endif()
endif()
